cd /home/rotsen/Projects/decibex

APP="decibex"
VERSION="0.1.0"
ARCH="amd64"
BUNDLE="build/linux/x64/release/bundle"
PKG="build/deb/${APP}_${VERSION}_${ARCH}"

test -d "$BUNDLE" || exit 1
test -x "$BUNDLE/DecibeX" || { echo "No se encuentra el ejecutable DecibeX"; exit 1; }

mkdir -p "$PKG/DEBIAN"
mkdir -p "$PKG/opt/decibex"
mkdir -p "$PKG/usr/share/applications"

cp -a "$BUNDLE/." "$PKG/opt/decibex/"

cat > "$PKG/DEBIAN/control" <<EOF
Package: decibex
Version: $VERSION
Section: sound
Priority: optional
Architecture: $ARCH
Maintainer: DecibeX
Description: Reproductor de musica DecibeX
 Reproductor de musica multiplataforma desarrollado con Flutter.
EOF

cat > "$PKG/usr/share/applications/decibex.desktop" <<EOF
[Desktop Entry]
Name=DecibeX
Comment=Reproductor de musica
Exec=/opt/decibex/DecibeX
Terminal=false
Type=Application
Categories=AudioVideo;Audio;Player;
EOF

dpkg-deb --root-owner-group --build "$PKG" "build/decibex_${VERSION}_${ARCH}.deb"
